cmake_minimum_required(VERSION 3.21)
project(qcurl_arch_component_contract LANGUAGES CXX)
add_compile_definitions(QT_NO_KEYWORDS)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(QCurl CONFIG REQUIRED PATHS "${QCURL_STAGE_PREFIX}" NO_DEFAULT_PATH)
foreach(target IN ITEMS QCurl::BlockingExtras QCurl::OtherExtras QCurl::TestSupport Qt6::Network)
    if(TARGET ${target})
        message(FATAL_ERROR "Default Core unexpectedly imports ${target}")
    endif()
endforeach()
get_target_property(core_links QCurl::QCurl INTERFACE_LINK_LIBRARIES)
if(NOT "Qt6::Core" IN_LIST core_links OR core_links MATCHES "CURL::|ZLIB::|libcurl_shared")
    message(FATAL_ERROR "Unexpected Core public dependencies: ${core_links}")
endif()

if(QCURL_REQUIRE_UNKNOWN)
    find_package(QCurl CONFIG REQUIRED COMPONENTS UnknownArchComponent
        PATHS "${QCURL_STAGE_PREFIX}" NO_DEFAULT_PATH)
else()
    find_package(QCurl CONFIG REQUIRED COMPONENTS Core BlockingExtras OtherExtras TestSupport
        PATHS "${QCURL_STAGE_PREFIX}" NO_DEFAULT_PATH)
    foreach(target IN ITEMS QCurl::QCurl QCurl::BlockingExtras QCurl::OtherExtras QCurl::TestSupport)
        if(NOT TARGET ${target})
            message(FATAL_ERROR "Complete component request is missing ${target}")
        endif()
    endforeach()
endif()

include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES QCurl::QCurl)
check_cxx_source_compiles("#include <QCNetworkReply_p.h>\nint main() { return 0; }" PRIVATE_HEADER_VISIBLE)
if(PRIVATE_HEADER_VISIBLE)
    message(FATAL_ERROR "Private header is visible in installed package")
endif()
